/* ================= ARTICLE PAGE (JSON-DRIVEN) ================= */ window.initArticlePage = async function () { try { const params = new URLSearchParams(window.location.search); const articleId = params.get("id"); if (!articleId) return; const res = await fetch("content/articles.json", { cache: "no-store" }); if (!res.ok) return; const data = await res.json(); const article = (data.items || []).find(a => a.id === articleId); if (!article) return; // Title & metadata document.getElementById("doc-title").textContent = article.title + " | The Mirror Jammu Kashmir"; document.getElementById("article-title").textContent = article.title; document.getElementById("article-category").textContent = article.category || article.section || ""; document.getElementById("article-author").textContent = article.author ? "By " + article.author : ""; document.getElementById("article-date").textContent = article.date ? " | " + article.date : ""; document.getElementById("article-readtime").textContent = article.readTime ? " | " + article.readTime : ""; // Image const img = document.getElementById("article-image"); if (img && article.image) { img.src = article.image; img.onerror = () => window.replaceWithPlaceholder(img); } // Content (summary expanded like Reuters intro) const body = document.getElementById("article-content"); body.innerHTML = ""; const lead = document.createElement("p"); lead.className = "article-lead"; lead.textContent = article.summary || ""; body.appendChild(lead); // Dummy paragraph expansion for now (safe default) const p = document.createElement("p"); p.textContent = "This report is part of The Mirror Jammu Kashmir’s ongoing coverage of political, human rights, and social developments. More verified updates will follow as information becomes available."; body.appendChild(p); // Init actions if (window.initPageActions) window.initPageActions(); if (window.initLikeButton) window.initLikeButton("article:" + article.id); } catch (e) { console.warn("Article load failed:", e); } }; /* ================= PAGE ACTIONS ================= */ window.initPageActions = function () { document.querySelectorAll("[data-copy]").forEach(btn => { btn.onclick = () => { navigator.clipboard.writeText(location.href); alert("Link copied"); }; }); document.querySelectorAll("[data-share]").forEach(btn => { btn.onclick = () => { if (navigator.share) { navigator.share({ title: document.title, url: location.href }); } else { navigator.clipboard.writeText(location.href); alert("Link copied"); } }; }); }; /* ================= LIKE SYSTEM (LOCAL SAFE) ================= */ window.initLikeButton = function (key) { const countEl = document.querySelector("[data-like-count]"); if (!countEl) return; const storageKey = "likes:" + key; let count = Number(localStorage.getItem(storageKey)) || 0; countEl.textContent = count; document.querySelectorAll("[data-like]").forEach(btn => { btn.onclick = () => { count++; localStorage.setItem(storageKey, count); countEl.textContent = count; }; }); };
